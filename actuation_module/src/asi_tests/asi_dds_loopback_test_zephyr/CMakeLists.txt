# asi_dds_loopback_test_zephyr: the test program actuation_module/test/dds_loopback_test.cpp
# as a Zephyr image. It is the entry of dds_loopback_test_bringup's Zephyr image
# (`[image.fvp-aemv8r-smp] entry = "asi_dds_loopback_test_zephyr"`): on Zephyr an
# owned entry is the west application itself, so this package finds Zephyr
# and `nros build` runs west on it. The FreeRTOS images build the same source
# through asi_dds_loopback_test, a plain CMake package; one package cannot be both.
#
# Its Kconfig is the image's os_config fragments (the controller's own, so the
# test runs on the configuration the controller ships with) plus the board
# network file; the capabilities and the RMW come from dds_loopback_test_bringup.

cmake_minimum_required(VERSION 3.22.1)

get_filename_component(_ACT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)

# asi_common's Kconfig symbols (CONFIG_LOG_LEVEL, CONFIG_NET_IFACE2_* and the
# rest it reads): it is a Zephyr module for its Kconfig only. nros build
# passes ZEPHYR_EXTRA_MODULES (nano-ros itself) on the command line, which
# Zephyr reads from the cache ahead of a local variable, so the module is
# appended to the cache entry. A generated controller application gets its
# workspace modules from nros build; an owned entry does not (yet).
set(_asi_modules ${ZEPHYR_EXTRA_MODULES} "${_ACT_ROOT}/src/asi_common")
list(REMOVE_DUPLICATES _asi_modules)
set(ZEPHYR_EXTRA_MODULES "${_asi_modules}" CACHE STRING "Zephyr modules outside the west manifest" FORCE)

find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(asi_dds_loopback_test_zephyr LANGUAGES C CXX)

find_package(nano_ros REQUIRED)

# The platform layer, without the network hook a generated controller entry
# calls: a test program's network is the board network file's, which Zephyr's
# net_config brings up before main().
set(ASI_COMMON_ENTRY_HOOK OFF CACHE BOOL "" FORCE)
if(NOT TARGET asi_common)
  add_subdirectory("${_ACT_ROOT}/src/asi_common" asi_common)
endif()

# The workspace's own message packages (safety_island_msgs,
# visionpilot_msgs), which asi_common's message umbrella includes.
set(NROS_INTERFACE_SEARCH_PATH "${_ACT_ROOT}/src")

# ROS2_MAIN: the program keeps its own main(); nano-ros renames it and calls
# it from the board's main(void). The interface packages package.xml names
# are generated by the verb. A west build names its image (NROS_IMAGE, which
# nros build passes) as the DEPLOY; the census's host build of this package
# (no `app` target) takes the host deploy.
set(_asi_deploy "")
if(TARGET app)
  set(_asi_deploy BOARD zephyr DEPLOY "${NROS_IMAGE}")
endif()
nano_ros_add_executable(asi_dds_loopback_test_zephyr "${_ACT_ROOT}/test/dds_loopback_test.cpp" ROS2_MAIN ${_asi_deploy})
target_link_libraries(asi_dds_loopback_test_zephyr PRIVATE asi_common)
